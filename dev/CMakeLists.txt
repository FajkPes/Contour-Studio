cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt" CONTOUR_APP_VERSION LIMIT_COUNT 1)
project(ContourStudio VERSION ${CONTOUR_APP_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(contour_core src/core/Contour.cpp src/core/Geometry.cpp src/core/Edges.cpp src/core/Compact.cpp src/core/Raster.cpp)
target_compile_definitions(contour_core PRIVATE NOMINMAX)
target_compile_definitions(contour_core PUBLIC CONTOUR_VERSION="${PROJECT_VERSION}")
target_include_directories(contour_core PUBLIC src)
if(MSVC)
  target_compile_options(contour_core PRIVATE /O2 /W4 /permissive- /utf-8)
else()
  target_compile_options(contour_core PRIVATE -O3 -Wall -Wextra)
endif()
add_executable(contour_tests tests/CoreTests.cpp)
target_link_libraries(contour_tests PRIVATE contour_core)
add_executable(contour_bench tools/Benchmark.cpp)
target_link_libraries(contour_bench PRIVATE contour_core)
if(WIN32)
  target_link_libraries(contour_core PUBLIC cabinet)
  target_sources(contour_bench PRIVATE src/app/ImageIO.cpp)
  target_link_libraries(contour_bench PRIVATE windowscodecs ole32)
  target_compile_definitions(contour_bench PRIVATE NOMINMAX)
endif()
enable_testing()
add_test(NAME core COMMAND contour_tests)
option(BUILD_WINUI "Build the Windows WinUI 3 application" OFF)
if(BUILD_WINUI)
  if(NOT WIN32 OR NOT DEFINED WINRT_HEADERS OR NOT DEFINED WINUI_PACKAGE OR NOT DEFINED FOUNDATION_PACKAGE)
    message(FATAL_ERROR "Use build.ps1 to locate Windows SDK and WinUI dependencies")
  endif()
  add_executable(ContourStudio WIN32 src/app/Main.cpp src/app/Capture.cpp src/app/ImageIO.cpp src/app/Renderer.cpp app.manifest)
  target_link_libraries(ContourStudio PRIVATE contour_core windowsapp d3d11 dxgi d2d1 windowscodecs comdlg32 shell32 shlwapi user32 ole32 "${FOUNDATION_PACKAGE}/lib/native/x64/Microsoft.WindowsAppRuntime.Bootstrap.lib")
  target_include_directories(ContourStudio PRIVATE "${WINRT_HEADERS}" "${WINUI_PACKAGE}/include" "${FOUNDATION_PACKAGE}/include")
  target_sources(contour_bench PRIVATE src/app/Renderer.cpp)
  target_compile_definitions(contour_bench PRIVATE CONTOUR_HAS_PREVIEW)
  target_include_directories(contour_bench PRIVATE "${WINUI_PACKAGE}/include")
  target_link_libraries(contour_bench PRIVATE d3d11 dxgi d2d1)
  target_compile_definitions(ContourStudio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_compile_options(ContourStudio PRIVATE /O2 /W4 /utf-8 /bigobj /permissive-)
  target_link_options(ContourStudio PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
  add_custom_command(TARGET ContourStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FOUNDATION_PACKAGE}/runtimes/win-x64/native/Microsoft.WindowsAppRuntime.Bootstrap.dll" "$<TARGET_FILE_DIR:ContourStudio>")
endif()
